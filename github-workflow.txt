# 云端打包 IPA —— 纯手机端操作，全程不需要电脑
#
# 用法：
#   1. 把整个 EmbyDanmaku 目录推到 GitHub 仓库
#   2. 打开仓库的 Actions 页面 → 选 "Build iOS IPA" → Run workflow
#   3. 选签名方式：
#        unsigned（默认）：产出未签名 IPA，装好 TrollStore 后直接永久安装
#        signed          ：需要付费开发者账号，产出已签名 IPA
#   4. 构建完成后自动发布到仓库的 Releases（标签 latest-ipa），
#      用 iPhone Safari 打开 Release 页面即可直接下载 IPA
#
# 默认完全不需要配置任何 Secrets。

name: Build iOS IPA

on:
  workflow_dispatch:
    inputs:
      sign_mode:
        description: '签名方式'
        required: true
        default: 'unsigned'
        type: choice
        options:
          - unsigned
          - signed
  push:
    tags:
      - 'v*'

permissions:
  contents: write

jobs:
  build:
    runs-on: macos-14

    env:
      PROJECT: EmbyDanmaku.xcodeproj
      TARGET: EmbyDanmaku
      APP_NAME: EmbyDanmaku
      # 签名模式下需要的 Secrets（unsigned 模式不需要）：
      #   P12_BASE64                 证书 .p12 的 base64
      #   P12_PASSWORD               导出 .p12 时设置的密码
      #   PROVISION_PROFILE_BASE64   .mobileprovision 的 base64
      #   CODE_SIGN_IDENTITY         例：Apple Development 或 iPhone Distribution
      #   TEAM_ID                    例：A1B2C3D4E5

    steps:
      - name: 检出代码
        uses: actions/checkout@v4

      - name: 显示 Xcode 版本
        run: |
          xcodebuild -version
          xcrun --sdk iphoneos --show-sdk-version

      # ---------------- 未签名构建（默认，零配置） ----------------
      - name: 构建未签名 .app
        if: ${{ github.event.inputs.sign_mode != 'signed' }}
        run: |
          set -o pipefail
          xcodebuild \
            -project "$PROJECT" \
            -target "$TARGET" \
            -configuration Release \
            -sdk iphoneos \
            SYMROOT=build OBJROOT=build/Intermediates \
            IPHONEOS_DEPLOYMENT_TARGET=15.0 \
            CODE_SIGNING_ALLOWED=NO \
            CODE_SIGNING_REQUIRED=NO \
            CODE_SIGN_IDENTITY="" \
            CODE_SIGN_STYLE=Manual \
            VALIDATE_PRODUCT=NO \
            GCC_WARN_INHIBIT_ALL_WARNINGS=YES \
            build | tail -30

      - name: 打包未签名 IPA
        if: ${{ github.event.inputs.sign_mode != 'signed' }}
        run: |
          APP="build/Release-iphoneos/$APP_NAME.app"
          if [ ! -d "$APP" ]; then
            echo "::error::未找到构建产物 $APP"
            ls -R build 2>/dev/null | head -40 || true
            exit 1
          fi
          mkdir -p Payload
          cp -R "$APP" Payload/
          zip -qry "${APP_NAME}-unsigned.ipa" Payload
          ls -lh "${APP_NAME}-unsigned.ipa"

      - name: 上传未签名 IPA（Artifacts，需登录 GitHub 才能下载）
        if: ${{ github.event.inputs.sign_mode != 'signed' }}
        uses: actions/upload-artifact@v4
        with:
          name: EmbyDanmaku-unsigned-ipa
          path: EmbyDanmaku-unsigned.ipa
          retention-days: 30

      # ---------------- 签名构建（需要付费开发者账号） ----------------
      - name: 导入签名证书与描述文件
        if: ${{ github.event.inputs.sign_mode == 'signed' }}
        env:
          P12: ${{ secrets.P12_BASE64 }}
          P12_PASSWORD: ${{ secrets.P12_PASSWORD }}
          PROFILE: ${{ secrets.PROVISION_PROFILE_BASE64 }}
        run: |
          if [ -z "$P12" ] || [ -z "$PROFILE" ]; then
            echo "::error::选择 signed 但未提供 P12_BASE64 / PROVISION_PROFILE_BASE64"
            exit 1
          fi
          echo "$P12" | base64 --decode > cert.p12
          echo "$PROFILE" | base64 --decode > app.mobileprovision
          security create-keychain -p "" build.keychain
          security default-keychain -s build.keychain
          security unlock-keychain -p "" build.keychain
          security set-keychain-settings -lut 21600 build.keychain
          security import cert.p12 -k build.keychain -P "$P12_PASSWORD" \
            -T /usr/bin/codesign -T /usr/bin/security
          security set-key-partition-list -S apple-tool:,apple:,codesign: -s -k "" build.keychain
          security list-keychains -d user -s build.keychain
          mkdir -p "$HOME/Library/MobileDevice/Provisioning Profiles"
          security cms -D -i app.mobileprovision > profile.plist
          UUID=$(/usr/libexec/PlistBuddy -c 'Print :UUID' profile.plist)
          PNAME=$(/usr/libexec/PlistBuddy -c 'Print :Name' profile.plist)
          echo "描述文件：$PNAME ($UUID)"
          cp app.mobileprovision "$HOME/Library/MobileDevice/Provisioning Profiles/$UUID.mobileprovision"
          echo "PROFILE_UUID=$UUID" >> $GITHUB_ENV
          echo "PROFILE_NAME=$PNAME" >> $GITHUB_ENV

      - name: 构建并签名 IPA
        if: ${{ github.event.inputs.sign_mode == 'signed' }}
        env:
          TEAM_ID: ${{ secrets.TEAM_ID }}
          CODE_SIGN_IDENTITY: ${{ secrets.CODE_SIGN_IDENTITY }}
        run: |
          set -o pipefail
          IDENTITY="${CODE_SIGN_IDENTITY:-Apple Development}"
          xcodebuild \
            -project "$PROJECT" \
            -target "$TARGET" \
            -configuration Release \
            -sdk iphoneos \
            SYMROOT=build-signed OBJROOT=build-signed/Intermediates \
            IPHONEOS_DEPLOYMENT_TARGET=15.0 \
            CODE_SIGNING_ALLOWED=YES \
            CODE_SIGNING_REQUIRED=YES \
            CODE_SIGN_IDENTITY="$IDENTITY" \
            DEVELOPMENT_TEAM="$TEAM_ID" \
            PROVISIONING_PROFILE_SPECIFIER="$PROFILE_NAME" \
            build | tail -30

          APP="build-signed/Release-iphoneos/$APP_NAME.app"
          if [ ! -d "$APP" ]; then
            echo "::error::未找到签名产物 $APP"
            exit 1
          fi
          codesign --verify --deep --strict "$APP"
          mkdir -p Payload
          cp -R "$APP" Payload/
          zip -qry "${APP_NAME}-signed.ipa" Payload
          ls -lh "${APP_NAME}-signed.ipa"

      - name: 上传已签名 IPA（Artifacts，需登录 GitHub 才能下载）
        if: ${{ github.event.inputs.sign_mode == 'signed' }}
        uses: actions/upload-artifact@v4
        with:
          name: EmbyDanmaku-signed-ipa
          path: EmbyDanmaku-signed.ipa
          retention-days: 30

      # ---------------- 发布到 Releases（手机 Safari 可直接下载，无需登录） ----------------
      - name: 发布到 Releases（标签 latest-ipa）
        if: success()
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          FILE=""
          [ -f "EmbyDanmaku-unsigned.ipa" ] && FILE="EmbyDanmaku-unsigned.ipa"
          [ -f "EmbyDanmaku-signed.ipa" ] && FILE="EmbyDanmaku-signed.ipa"
          if [ -z "$FILE" ]; then
            echo "没有可发布的 IPA，跳过"
            exit 0
          fi
          NOTES="自动构建于 $(date -u '+%Y-%m-%d %H:%M') UTC · 最低系统 iOS 15.0
          未签名 IPA 推荐用 TrollStore 安装（iOS 14.0 ~ 16.6.1，永久免签不掉签）；
          也可用 Feather / Scarlet / Ksign 等手机端签名工具安装（免费证书 7 天有效）。"
          gh release delete latest-ipa --yes --cleanup-tag 2>/dev/null || true
          gh release create latest-ipa \
            --prerelease \
            --title "弹幕影院 · 最新构建" \
            --notes "$NOTES" \
            "$FILE"
          echo "已发布：$(gh release view latest-ipa --json url -q .url)"
